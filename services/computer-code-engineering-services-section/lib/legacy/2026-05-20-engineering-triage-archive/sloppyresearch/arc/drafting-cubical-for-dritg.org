{-# OPTIONS --cubical --safe #-}

module DRITG.Core.Equiv where

open import Cubical.Core.Everything
open import Cubical.Foundations.Everything
open import Cubical.Foundations.Equiv
  renaming (_≃_ to _≃_; equivFun to fun; equivInv to inv)

-- Universe-polymorphic Sets
variable
  ℓ ℓ₁ ℓ₂ ℓ₃ : Level

------------------------------------------------------------------------
-- Identity, composition, inversion of equivalences
------------------------------------------------------------------------

id≃ : ∀ {A : Set ℓ} → A ≃ A
id≃ = record
  { fun      = λ x → x
  ; inv      = λ x → x
  ; leftInv  = λ _ → refl
  ; rightInv = λ _ → refl
  }

infixr 60 _∘e_
_∘e_ : ∀ {A : Set ℓ₁} {B : Set ℓ₂} {C : Set ℓ₃}
     → (A ≃ B) → (B ≃ C) → (A ≃ C)
_∘e_ e₁ e₂ =
  record
    { fun      = (fun e₂) ∘ (fun e₁)
    ; inv      = (inv e₁) ∘ (inv e₂)
    ; leftInv  = λ a → cong (fun e₂) (leftInv e₁ a) ∙ leftInv e₂ (fun e₁ a)
    ; rightInv = λ c → cong (inv e₁) (rightInv e₂ c) ∙ rightInv e₁ (inv e₂ c)
    }

sym≃ : ∀ {A : Set ℓ₁} {B : Set ℓ₂} → (A ≃ B) → (B ≃ A)
sym≃ e = record
  { fun      = inv e
  ; inv      = fun e
  ; leftInv  = rightInv e
  ; rightInv = leftInv  e
  }

------------------------------------------------------------------------
-- Basic “algebraic laws” at the granularity we actually need
-- (They’re stated as *equivalences* between equivalences, which is
-- stronger than a mere Path and good enough for normalization.)
------------------------------------------------------------------------

assoc≃ : ∀ {A : Set ℓ₁} {B : Set ℓ₂} {C : Set ℓ₃} {D : Set (ℓ₁ ⊔ ℓ₂ ⊔ ℓ₃)}
        (e₁ : A ≃ B) (e₂ : B ≃ C) (e₃ : C ≃ D)
      → (e₁ ∘e e₂) ∘e e₃ ≃ e₁ ∘e (e₂ ∘e e₃)
assoc≃ _ _ _ = id≃

id-left≃  : ∀ {A : Set ℓ₁} {B : Set ℓ₂} (e : A ≃ B) → id≃ ∘e e ≃ e
id-left≃ _ = id≃

id-right≃ : ∀ {A : Set ℓ₁} {B : Set ℓ₂} (e : A ≃ B) → e ∘e id≃ ≃ e
id-right≃ _ = id≃

sym-sym≃  : ∀ {A : Set ℓ₁} {B : Set ℓ₂} (e : A ≃ B) → sym≃ (sym≃ e) ≃ e
sym-sym≃ _ = id≃

------------------------------------------------------------------------
-- Homotopies, squares, and whiskering (for coherence packaging)
------------------------------------------------------------------------

-- A homotopy is a path in function space.
Homotopy : ∀ {A : Set ℓ₁} {B : Set ℓ₂} → (A → B) → (A → B) → Set (ℓ₁ ⊔ ℓ₂)
Homotopy f g = Path (A → B) f g

-- Minimal square type carrying the edge (homotopy).
record Square {A : Set ℓ₁} {B : Set ℓ₂} (f g : A → B) : Set (ℓ₁ ⊔ ℓ₂) where
  field edge : Path (A → B) f g
open Square public

-- Apply a path in function space to a point (pointwise rewrite).
applyPath : ∀ {A : Set ℓ₁} {B : Set ℓ₂} {f g : A → B}
          → Path (A → B) f g → (a : A) → Path B (f a) (g a)
applyPath p a i = p i a

-- Left/right whiskering of homotopies (very handy for coherence proofs).
whiskerL : ∀ {A : Set ℓ₁} {B : Set ℓ₂} {C : Set ℓ₃}
         → (h : B → C) → {f g : A → B} → Homotopy f g → Homotopy (h ∘ f) (h ∘ g)
whiskerL h p i a = h (p i a)

whiskerR : ∀ {A : Set ℓ₁} {B : Set ℓ₂} {C : Set ℓ₃}
         → {h k : B → C} → Homotopy h k → (f : A → B) → Homotopy (h ∘ f) (k ∘ f)
whiskerR p f i a = p i (f a)
