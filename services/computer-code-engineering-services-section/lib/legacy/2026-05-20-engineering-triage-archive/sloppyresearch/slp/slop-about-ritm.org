"""
cell_rules.py

Each function corresponds directly to a modality or coupling operation
from the formal RITM structure:

    □   — "box" : reflexive/transitive closure on Y
    ◇   — "diamond" : closure on the transpose of Y
    C   — "composite" : ◇ followed by □
    δ   — "delta" : local fibre lift creating higher-level targets
    RC  — relational coupling propagation (x∼a & a→b ⇒ x∼b)
    RT  — relational imprint: rows of R yield cliques in Y

These laws are imported by `cell.py`, which instantiates a concrete
RelationalCell object that enacts them dynamically.
"""

from __future__ import annotations
from typing import Dict, Set, Tuple, Optional

# ----------------------------------------------------------------------------- 
# --- Type definitions --------------------------------------------------------
# -----------------------------------------------------------------------------

Node = Tuple[int, int]       # (identifier, fibre-level)
YAdj = Dict[Node, Set[Node]] # directed adjacency (Y: morphisms)
RRel = Dict[Node, Set[Node]] # symmetric relational coupling (R: relations)
YEvent = Tuple[Node, Node]   # proposed arrow addition
REvent = Tuple[Node, Node]   # proposed relation addition


# ----------------------------------------------------------------------------- 
# --- Core utilities ----------------------------------------------------------
# -----------------------------------------------------------------------------

def nodeset(Y: YAdj) -> Set[Node]:
    """Return all nodes appearing in Y (sources or targets)."""
    s = set(Y.keys())
    for js in Y.values():
        s |= js
    return s


def transpose_Y(Y: YAdj) -> YAdj:
    """Return the transpose of Y (reverse all arrows)."""
    T: YAdj = {n: set() for n in nodeset(Y)}
    for i, js in Y.items():
        for j in js:
            T.setdefault(j, set()).add(i)
    return T


# ----------------------------------------------------------------------------- 
# --- Modal laws --------------------------------------------------------------
# -----------------------------------------------------------------------------

def box(Y: YAdj) -> Optional[YEvent]:
    """
    □ — Reflexive/transitive closure on Y.
    Returns the next missing reflexive or transitive edge, if any.
    """
    ns = sorted(nodeset(Y))
    # Reflexivity: add (a,a) if missing
    for a in ns:
        if a not in Y or a not in Y[a]:
            return (a, a)
    # Transitivity: add (i,k) if i→j and j→k but not i→k
    for i in ns:
        for j in sorted(Y.get(i, ())):
            for k in sorted(Y.get(j, ())):
                if k not in Y.get(i, ()):
                    return (i, k)
    return None


def diamond(Y: YAdj) -> Optional[YEvent]:
    """
    ◇ — Closure computed on the transpose of Y, mirrored back into Y.

    Rules on the transpose T:
        D2: a -> a      (reflexivity)
        D1: j->k & k->i ⇒ j->i
    Mirrored back to Y as (i -> j).
    """
    T = transpose_Y(Y)
    ns = sorted(nodeset(T))

    # Reflexivity on T ⇒ (a,a) in Y
    for a in ns:
        if a not in Y or a not in Y.get(a, ()):
            return (a, a)

    # Transitivity on T ⇒ (i->j) in Y
    for j in ns:
        for k in sorted(T.get(j, ())):
            for i in sorted(T.get(k, ())):
                if j not in Y.get(i, ()):
                    return (i, j)
    return None


def composite(Y: YAdj) -> Optional[YEvent]:
    """C = □∘◇ — compute ◇ first, then □."""
    d = diamond(Y)
    if d:
        return d
    return box(Y)


def delta(Y: YAdj) -> Optional[YEvent]:
    """
    δ — Local-fibre lift.
    For each i→j, propose i→(j_id, j_lvl+1) if missing.
    Ensures indefinite expansion (non-stabilization).
    """
    for i, js in list(Y.items()):
        for j in list(js):
            lifted = (j[0], j[1] + 1)
            if lifted not in Y.get(i, ()):
                return (i, lifted)
    return None


# ----------------------------------------------------------------------------- 
# --- Coupling and imprint laws ----------------------------------------------
# -----------------------------------------------------------------------------

def RC(R: RRel, e: YEvent) -> Optional[REvent]:
    """
    RC — Relational coupling propagation.
    If x∼a and a→b, propose x∼b.
    """
    a, b = e
    for x, ys in R.items():
        if a in ys and b not in ys:
            return (x, b)
    return None


def RT(Y: YAdj, row: Set[Node]) -> Optional[YEvent]:
    """
    RT — Relational imprint.
    Given a row (set of related nodes) from R,
    propose one missing oriented edge i→j between its members.
    """
    for i in row:
        for j in row:
            if j not in Y.get(i, ()):
                return (i, j)
    return None


# ----------------------------------------------------------------------------- 
# --- Seeding convenience -----------------------------------------------------
# -----------------------------------------------------------------------------

def seed_one_arrow(i_id: int, j_id: int) -> Tuple[YAdj, RRel]:
    """
    Construct initial Y and R with a single arrow i→j and symmetric relation i∼j.
    Used for bootstrapping the minimal relational cell.
    """
    i = (i_id, 0)
    j = (j_id, 0)
    Y: YAdj = {i: {j}}
    R: RRel = {i: {j}, j: {i}}
    return Y, R

"""
cell.py — Constructive Relational ∞-Topos Cell

Implements the *RelationalCell* object: the minimal constructive model
of the Relational ∞-Topos Modality (RITM) described in the formal document.

This class provides a mutable presheaf cell (Y,R) that enacts the
axiomatic laws defined in `cell_laws.py`.  Each event step applies
one modality or coupling rule (□,◇,C,δ,RC,RT), progressively generating
a self-stabilizing relational field.

Conceptual layering:
    cell_laws.py   — defines the pure axioms (stateless inference)
    cell.py        — enacts those axioms (constructive presheaf model)
    category.py    — formalizes them as functors, adjunctions, monads

Author: em senn
Date: 2025-10-17
"""

from __future__ import annotations
from typing import Optional, Tuple, List

from cell_rules import (
    Node, YAdj, RRel, YEvent, REvent,
    seed_one_arrow,
    box, diamond, composite, delta,

class RelationalCell:
    """
    The constructive presheaf cell implementing the Relational ∞-Topos Modality.

    Attributes:
        Y : dict[node, set[node]]
            Directed adjacency relation (spatial arrows)
        R : dict[node, set[node]]
            Symmetric relational coupling
    """

    def __init__(self, i_id: int = 0, j_id: int = 1):
        """Initialize cell with a single seed arrow i→j."""
        self.Y, self.R = seed_one_arrow(i_id, j_id)
        self.history: List[Tuple[str, Tuple[Node, Node]]] = []

    # -------------------------------------------------------------------------
    # --- Core dynamics -------------------------------------------------------
    # -------------------------------------------------------------------------

    def step(self, prefer: str = "Cδ") -> Optional[Tuple[str, Tuple[Node, Node]]]:
        """
        Execute one event according to the prioritized constructive rules.

        Order:
            1. C (composite = ◇ then □)
            2. RT (relational imprint)
            3. δ (local fibre lift)
        After any Y-event, apply one RC (relational coupling).

        Returns:
            (kind, event) if a new event occurs, or None if stable.
        """
        evt: Optional[YEvent] = None
        kind: Optional[str] = None

        # Step 1: composite (C)
        if prefer.startswith("C"):
            evt = composite(self.Y)
            kind = "C"

        # Step 2: relational imprint (RT)
        if evt is None:
            for row in self.R.values():
                evt = RT(self.Y, row)
                if evt:
                    kind = "RT"
                    break

        # Step 3: delta-lift (δ)
        if evt is None and "δ" in prefer:
            evt = delta(self.Y)
            kind = "δ"

        # Apply event if found
        if evt:
            i, j = evt
            self.Y.setdefault(i, set()).add(j)

            # RC coupling after any Y-edge
            rc_evt = RC(self.R, evt)
            if rc_evt:
                a, b = rc_evt
                self.R.setdefault(a, set()).add(b)
                self.R.setdefault(b, set()).add(a)

            self.history.append((kind, evt))
            return kind, evt

        return None

    # -------------------------------------------------------------------------
    # --- Iterative dynamics --------------------------------------------------
    # -------------------------------------------------------------------------

    def run(self, steps: int = 1_000, prefer: str = "Cδ") -> None:
        """
        Run successive step() calls until stability or step limit reached.
        """
        for _ in range(steps):
            event = self.step(prefer=prefer)
            if event is None:
                break

    # -------------------------------------------------------------------------
    # --- Inspection utilities ------------------------------------------------
    # -------------------------------------------------------------------------

    def node_count(self) -> int:
        """Return total number of distinct nodes present in Y."""
        return len({n for src, dsts in self.Y.items() for n in [src, *dsts]})

    def edge_count(self) -> int:
        """Return total number of directed edges in Y."""
        return sum(len(dsts) for dsts in self.Y.values())

    def relation_count(self) -> int:
        """Return total number of symmetric relations in R."""
        return sum(len(v) for v in self.R.values()) // 2

    def summary(self) -> str:
        """Return a brief textual summary of current cell state."""
        return (
            f"RelationalCell: {self.node_count()} nodes, "
            f"{self.edge_count()} edges, "
            f"{self.relation_count()} relations"
        )

    def __repr__(self) -> str:
        return self.summary()

"""
category.py — Categorical wrapper for the Constructive Relational ∞-Topos Cell

This module lifts the constructive cell (Y,R) into categorical structure:

- Category  E:  objects are finite presheaf cells (Y,R)

- Naturals: natural transformations with brute-force naturality checks
- Adjunction: unit/counit + triangle identity checkers on finite test sets

Design goals:
- Keep it minimal and testable on finite instances.
- Represent the RITM chain via object-level saturations that are idempotent.
- Make room to encode units/counits explicitly in later passes.

@dataclass(frozen=True)
class RelationalObject:
    """A finite presheaf cell (Y,R) viewed as an object in E."""
    Y: Mapping[Node, Set[Node]]
    R: Mapping[Node, Set[Node]]

    def nodes(self) -> Set[Node]:
        s: Set[Node] = set(self.Y.keys())
        for js in self.Y.values(): s |= js
        return s


@dataclass(frozen=True)
class RelationalMorphism:
    """
    A morphism f : X -> Y is a node-map that preserves:
      - directed edges: i→j in X.Y implies f(i)→f(j) in Y.Y
      - symmetric relations: i∼j in X.R implies f(i)∼f(j) in Y.R
    """
    src: RelationalObject
    tgt: RelationalObject
    f: Mapping[Node, Node]

    def is_edge_preserving(self) -> bool:
        Ysrc, Ytgt = self.src.Y, self.tgt.Y
        for i, js in Ysrc.items():
            fi = self.f.get(i)
            if fi is None: return False
            for j in js:
                fj = self.f.get(j)
                if fj is None: return False
                if fj not in Ytgt.get(fi, ()):
                    return False
        return True

    def is_relation_preserving(self) -> bool:
        Rsrc, Rtgt = self.src.R, self.tgt.R
        for i, js in Rsrc.items():
            fi = self.f.get(i)
            if fi is None: return False
            for j in js:
                fj = self.f.get(j)
                if fj is None: return False
                if fi not in Rtgt.get(fj, ()) or fj not in Rtgt.get(fi, ()):
                    return False
        return True

    def is_valid(self) -> bool:
        return self.is_edge_preserving() and self.is_relation_preserving()


# =============================================================================
# Category + composition
# =============================================================================

class Category:
    """Very small helper for composing morphisms and holding objects."""
    def __init__(self) -> None:
        self.objects: List[RelationalObject] = []

    def add_object(self, obj: RelationalObject) -> RelationalObject:
        self.objects.append(obj)
        return obj

    @staticmethod
    def id(obj: RelationalObject) -> RelationalMorphism:
        f = {n: n for n in obj.nodes()}
        return RelationalMorphism(obj, obj, f)

    @staticmethod
    def compose(g: RelationalMorphism, f: RelationalMorphism) -> RelationalMorphism:
        """g ∘ f (requires f.tgt == g.src)."""
        if f.tgt != g.src:
            raise ValueError("Cannot compose: codomain/domain mismatch")
        comp = {k: g.f[f.f[k]] for k in f.f.keys()}
        h = RelationalMorphism(f.src, g.tgt, comp)
        if not h.is_valid():
            raise ValueError("Composite is not a valid morphism in E")
        return h

@dataclass(frozen=True)
class Functor:
    """Endofunctor F : E -> E with object/map actions and sanity checks."""
    on_obj: Callable[[RelationalObject], RelationalObject]
    on_mor: Callable[[RelationalMorphism], RelationalMorphism]
    name: str = "F"

    def map_obj(self, X: RelationalObject) -> RelationalObject:
        return self.on_obj(X)

    def map_mor(self, f: RelationalMorphism) -> RelationalMorphism:
        return self.on_mor(f)

    def __matmul__(self, G: "Functor") -> "Functor":
        """(self @ G)(-) = self(G(-)), i.e., functor composition."""
        def comp_obj(X: RelationalObject) -> RelationalObject:
            return self.on_obj(G.on_obj(X))
        def comp_mor(f: RelationalMorphism) -> RelationalMorphism:
            return self.on_mor(G.on_mor(f))
        return Functor(comp_obj, comp_mor, name=f"{self.name}∘{G.name}")


# --- Saturation helpers ------------------------------------------------------

def _copy_Y(Y: Mapping[Node, Set[Node]]) -> YAdj:
    return {i: set(js) for i, js in Y.items()}

def _copy_R(R: Mapping[Node, Set[Node]]) -> RRel:
    return {i: set(js) for i, js in R.items()}

def saturate_Y_via_step(
    X: RelationalObject,
    step_fn: Callable[[YAdj], Optional[Tuple[Node, Node]]],
    max_steps: int = 1000
) -> RelationalObject:
    """Repeatedly apply a single-step Y-law up to max_steps."""
    Y = _copy_Y(X.Y)
    for _ in range(max_steps):
        evt = step_fn(Y)
        if not evt:
            break
        i, j = evt
        Y.setdefault(i, set()).add(j)
    return RelationalObject(Y=Y, R=_copy_R(X.R))

def saturate_delta(
    X: RelationalObject,
    max_steps: int = 100
) -> RelationalObject:
    """Apply δ repeatedly but stop after max_steps."""
    Y = _copy_Y(X.Y)
    R = _copy_R(X.R)
    for _ in range(max_steps):
        evt = delta(Y)
        if not evt:
            break
        i, j = evt
        Y.setdefault(i, set()).add(j)
        rc_evt = RC(R, evt)
        if rc_evt:
            a, b = rc_evt
            R.setdefault(a, set()).add(b)
            R.setdefault(b, set()).add(a)
    return RelationalObject(Y=Y, R=R)


def lift_morphism_pointwise(F_on_obj: Callable[[RelationalObject], RelationalObject]
                            ) -> Callable[[RelationalMorphism], RelationalMorphism]:
    """
    Given an object-action that returns a new (Y,R),
    push the same node-map through and validate preservation.
    """
    def on_mor(f: RelationalMorphism) -> RelationalMorphism:
        src2 = F_on_obj(f.src)
        tgt2 = F_on_obj(f.tgt)
        h = RelationalMorphism(src2, tgt2, dict(f.f))
        if not h.is_valid():
            raise ValueError("Image of morphism under functor is not valid")
        return h
    return on_mor

@dataclass
class NaturalTransformation:
    """
    α : F ⇒ G with components α_X : F(X) → G(X) as RelationalMorphisms.
    Naturalities are checked against provided test morphisms.
    """
    F: Functor
    G: Functor
    components: Dict[RelationalObject, RelationalMorphism]
    name: str = "α"

    def check_naturality(self, test_morphisms: Iterable[RelationalMorphism]) -> bool:
        ok = True
        for f in test_morphisms:
            FX  = self.F.map_obj(f.src)
            FY  = self.F.map_obj(f.tgt)
            GX  = self.G.map_obj(f.src)
            GY  = self.G.map_obj(f.tgt)
            alpha_X = self.components.get(f.src)
            alpha_Y = self.components.get(f.tgt)
            if alpha_X is None or alpha_Y is None:
                ok = False; break
            # G(f) ∘ α_X  ==  α_Y ∘ F(f)
            Gf = self.G.map_mor(f)
            Ff = self.F.map_mor(f)
            left  = RelationalMorphism(FX, GY, {n: Gf.f[alpha_X.f[n]] for n in alpha_X.f})
            right = RelationalMorphism(FX, GY, {n: alpha_Y.f[Ff.f[n]] for n in Ff.f})
            if left.f != right.f:
                ok = False; break
        return ok


@dataclass
class Adjunction:
    """
    (F ⊣ G) with unit η : Id ⇒ G∘F and counit ε : F∘G ⇒ Id,
    plus triangle identity checkers on finite tests.
    """
    F: Functor
    G: Functor
    unit: NaturalTransformation
    counit: NaturalTransformation
    name: str = "F⊣G"

    def check_triangles(self,
                        E: Category,
                        test_objs: Iterable[RelationalObject]) -> bool:
        ok = True
        for X in test_objs:
            # F --Fη_X--> FGF --ε_FX--> F  equals id_F
            FX   = self.F.map_obj(X)
            GFX  = self.G.map_obj(FX)
            FGFx = self.F.map_obj(GFX)
            # Compose as maps on node-sets; here we just compare dict composition
            eta_X = self.unit.components.get(X)
            eps_FX = self.counit.components.get(FX)
            if eta_X is None or eps_FX is None:
                ok = False; break
            comp = {n: eps_FX.f.get(eta_X.f.get(n, n), n) for n in FX.nodes()}
            idF = {n: n for n in FX.nodes()}
            if comp != idF:
                ok = False; break

            # Dually: G --η_GX--> GFG --Gε_X--> G equals id_G
            GX   = self.G.map_obj(X)
            FGX  = self.F.map_obj(GX)
            GFGx = self.G.map_obj(FGX)
            eta_GX = self.unit.components.get(GX)
            eps_X  = self.counit.components.get(X)
            if eta_GX is None or eps_X is None:
                ok = False; break
            # Build G(ε_X) action on node maps by mapping morphisms through G
            try:
                epsX_mor = RelationalMorphism(self.F.map_obj(self.G.map_obj(X)), X, {n: n for n in GFGx.nodes()})  # placeholder shape
                Gepx = self.G.map_mor(epsX_mor)  # may raise if invalid
            except Exception:

"""
geometry.py — Cohesive geometric reflection of the Relational ∞-Topos Cell

This layer lifts categorical RITM structure into a *geometric* one,
interpreting relational cells as local charts of smooth cohesive spaces.

Each RelationalObject becomes a *geometric chart* G(X):
    - nodes -> points in a local manifold patch
    - Y edges -> differentiable structure (flows)
    - R relations -> metric / connection constraints

Functorial structure:
    Π→   : geometric 'shape' functor (integration along path)
    □    : 'flat' space of paths  (Map(I_d, X))
    ◇    : 'sharp' dual path space (Map(X, I_d))
    ∇←   : 'codiscrete' / global section functor (limit)
    δgeo : local lift of fibres (connection / differential)

All are endofunctors on the category of GeometricCells.
"""

from __future__ import annotations
from dataclasses import dataclass
from typing import Callable, Dict, Set, Tuple, List
import math

from category import (
    RelationalObject,
    Functor,
    BOX, DIAMOND, COMPOSITE, DELTA
)

# ---------------------------------------------------------------------
# --- Geometric primitives --------------------------------------------
# ---------------------------------------------------------------------

@dataclass(frozen=True)
class Point:
    """A smooth-space point with coordinates (x,y,level)."""
    x: float
    y: float
    level: int = 0

@dataclass
class GeometricCell:
    """
    Geometric realization of a relational object.
    Points correspond to nodes; edges become local vector fields.
    """
    points: Dict[Tuple[int, int], Point]
    edges: Set[Tuple[Tuple[int,int], Tuple[int,int]]]
    relations: Set[Tuple[Tuple[int,int], Tuple[int,int]]]

    def as_relational_object(self) -> RelationalObject:
        """Forget geometry -> return underlying combinatorial cell."""
        Y = {}
        for (i,j) in self.edges:
            Y.setdefault(i, set()).add(j)
        R = {}
        for (i,j) in self.relations:
            R.setdefault(i, set()).add(j)
            R.setdefault(j, set()).add(i)
        return RelationalObject(Y=Y, R=R)

# ---------------------------------------------------------------------
# --- Smooth-space transforms -----------------------------------------
# ---------------------------------------------------------------------

def shape(cell: GeometricCell) -> GeometricCell:
    """
    Π→ : Shape functor — collapse paths to endpoints (integration along I_d).
    In discrete terms: retain points but remove internal edges.
    """
    pts = dict(cell.points)
    # Remove interior edges (those forming 2-step chains)
    new_edges = set()
    for (i,j) in cell.edges:
        if all(k not in cell.edges for k in [(i,j)]):
            new_edges.add((i,j))
    return GeometricCell(points=pts, edges=new_edges, relations=set(cell.relations))

def flat(cell: GeometricCell) -> GeometricCell:
    """
    □ : Flat functor — space of paths (Map(I_d, X)).
    Duplicate points along a synthetic time axis to simulate tangent bundle.
    """
    pts = {}
    for (nid, lvl), p in cell.points.items():
        pts[(nid, lvl)] = Point(p.x, p.y, p.level)
        pts[(nid, lvl+1)] = Point(p.x + 0.0, p.y + 0.0, p.level + 1)
    edges = set(cell.edges)
    # Add vertical lifts
    for (nid, lvl) in list(pts.keys()):
        lifted = (nid, lvl+1)
        if lifted in pts:
            edges.add(((nid,lvl), lifted))
    return GeometricCell(points=pts, edges=edges, relations=set(cell.relations))

def sharp(cell: GeometricCell) -> GeometricCell:
    """
    ◇ : Sharp functor — dual path space (Map(X, I_d)).
    Transpose edges (reverse orientation).
    """
    pts = dict(cell.points)
    edges = {(j,i) for (i,j) in cell.edges}
    return GeometricCell(points=pts, edges=edges, relations=set(cell.relations))

def codiscrete(cell: GeometricCell) -> GeometricCell:
    """
    ∇← : Codiscrete functor — global section / limit.
    Every point connected to every other (discrete global binding).
    """
    pts = dict(cell.points)
    all_pairs = {(i,j) for i in pts for j in pts if i != j}
    return GeometricCell(points=pts, edges=all_pairs, relations=all_pairs)

def delta_geo(cell: GeometricCell) -> GeometricCell:
    """
    δ_geo : Geometric fibre lift (connection).
    Adds small offset points in tangent direction to simulate differentiation.
    """
    pts = dict(cell.points)
    new_pts = {}
    for (nid, lvl), p in pts.items():
        lifted = (nid, lvl+1)
        new_pts[lifted] = Point(p.x + 0.1, p.y + 0.1, p.level + 1)
    pts.update(new_pts)
    edges = set(cell.edges)
    edges |= {((nid, lvl), (nid, lvl+1)) for (nid, lvl) in pts}
    return GeometricCell(points=pts, edges=edges, relations=set(cell.relations))

# ---------------------------------------------------------------------
# --- Cohesive functors ------------------------------------------------
# ---------------------------------------------------------------------

PI = Functor(on_obj=lambda X: shape(X), on_mor=lambda f: f, name="Π")
BOX_GEO = Functor(on_obj=lambda X: flat(X), on_mor=lambda f: f, name="□geo")
DIAMOND_GEO = Functor(on_obj=lambda X: sharp(X), on_mor=lambda f: f, name="◇geo")
NABLA = Functor(on_obj=lambda X: codiscrete(X), on_mor=lambda f: f, name="∇")
DELTA_GEO = Functor(on_obj=lambda X: delta_geo(X), on_mor=lambda f: f, name="δgeo")

# ---------------------------------------------------------------------
# --- Conversion utilities --------------------------------------------
# ---------------------------------------------------------------------

def from_relational_object(X: RelationalObject) -> GeometricCell:
    """Embed a RelationalObject as a geometric chart with planar coordinates."""
    pts = {}
    for (nid,lvl) in X.Y.keys():
        pts[(nid,lvl)] = Point(float(nid), float(lvl))
    for js in X.Y.values():
        for (nid,lvl) in js:
            if (nid,lvl) not in pts:
                pts[(nid,lvl)] = Point(float(nid), float(lvl))
    edges = {(i,j) for i,js in X.Y.items() for j in js}
    relations = {(i,j) for i,js in X.R.items() for j in js}
    return GeometricCell(points=pts, edges=edges, relations=relations)

def visualize_ascii(cell: GeometricCell) -> str:
    """
    Very minimal ASCII layout: prints node levels and connectivity.
    """
    lines = []
    for (nid,lvl), p in sorted(cell.points.items()):
        targets = [str(j) for (i,j) in cell.edges if i==(nid,lvl)]
        lines.append(f"{(nid,lvl)} -> {targets}")
    return "\n".join(lines)

"""
physics.py — Field dynamics over cohesive geometric cells

Builds on:
    geometry.py  (Π, □geo, ◇geo, ∇, δgeo and GeometricCell/Point)

Capabilities:
- Node-wise scalar/vector fields on a GeometricCell

- Simple time-steppers (Euler, semi-implicit / symplectic-ish)
- Energy functionals (kinetic, potential) + quick invariant checks
- Pullback/pushforward of fields along cohesive functors

Design notes:
- Keeps everything finite and terminating; no infinite δ lifting loops.
- Uses immutable Point replacement to avoid in-place geometry skew.

Author: em senn
Date: 2025-10-17
"""

from __future__ import annotations
from dataclasses import dataclass, replace
from typing import Dict, Tuple, Callable, Optional, Iterable
import math

from geometry import (
    GeometricCell, Point,
    PI, BOX_GEO, DIAMOND_GEO, NABLA, DELTA_GEO,
    from_relational_object, visualize_ascii
)
from category import seed_object

Node = Tuple[int, int]
Scalar = float
Vec2 = Tuple[float, float]

@dataclass(frozen=True)
class ScalarField:
    """Node-wise scalar field φ : nodes → ℝ."""
    data: Dict[Node, Scalar]

    def pullback(self, F: Callable[[GeometricCell], GeometricCell],
                 src: GeometricCell) -> "ScalarField":
        """
        Pull φ back along F: F*φ defined on nodes of F(src) by nearest-node copy.
        (Minimal, discrete approximation.)
        """
        tgt = F(src)
        out = {}
        for n, p in tgt.points.items():
            # choose nearest original node in src (Euclidean)
            cn = min(src.points.items(),
                     key=lambda kv: (kv[1].x - p.x)**2 + (kv[1].y - p.y)**2)[0]
            out[n] = self.data.get(cn, 0.0)
        return ScalarField(out)


@dataclass(frozen=True)
class VectorField:
    """Node-wise vector field v : nodes → ℝ^2."""
    data: Dict[Node, Vec2]

    def pullback(self, F: Callable[[GeometricCell], GeometricCell],
                 src: GeometricCell) -> "VectorField":
        """
        Pull v back along F (nearest-node copy, keeping the vector).
        """
        tgt = F(src)
        out = {}
        for n, p in tgt.points.items():
            cn = min(src.points.items(),
                     key=lambda kv: (kv[1].x - p.x)**2 + (kv[1].y - p.y)**2)[0]
            out[n] = self.data.get(cn, (0.0, 0.0))
        return VectorField(out)


# -----------------------------------------------------------------------------
# Discrete differential operators
# -----------------------------------------------------------------------------

def neighbors(cell: GeometricCell, n: Node) -> Iterable[Node]:
    for (i, j) in cell.edges:
        if i == n:
            yield j

def gradient(cell: GeometricCell, phi: ScalarField, eps: float = 1e-9) -> VectorField:
    """
    Discrete (outgoing) gradient: grad φ (n) ≈ avg over neighbors (φ(j)-φ(n)) * unit(n→j).
    """
    out: Dict[Node, Vec2] = {}
    for n, p in cell.points.items():
        accx = accy = 0.0
        cnt = 0
        for j in neighbors(cell, n):
            pj = cell.points.get(j)
            if pj is None:
                continue
            dx = pj.x - p.x
            dy = pj.y - p.y
            dist = math.hypot(dx, dy) + eps
            u = (dx / dist, dy / dist)
            accx += (phi.data.get(j, 0.0) - phi.data.get(n, 0.0)) * u[0]
            accy += (phi.data.get(j, 0.0) - phi.data.get(n, 0.0)) * u[1]
            cnt += 1
        if cnt > 0:
            out[n] = (accx / cnt, accy / cnt)
        else:
            out[n] = (0.0, 0.0)
    return VectorField(out)

def divergence(cell: GeometricCell, v: VectorField, eps: float = 1e-9) -> ScalarField:
    """
    Discrete divergence: div v (n) ≈ avg over neighbors ⟨v(j)-v(n), unit(n→j)⟩.
    """
    out: Dict[Node, float] = {}
    for n, p in cell.points.items():
        vn = v.data.get(n, (0.0, 0.0))
        acc = 0.0
        cnt = 0
        for j in neighbors(cell, n):
            pj = cell.points.get(j)
            if pj is None:
                continue
            dx = pj.x - p.x
            dy = pj.y - p.y
            dist = math.hypot(dx, dy) + eps
            u = (dx / dist, dy / dist)
            vj = v.data.get(j, (0.0, 0.0))
            acc += (vj[0] - vn[0]) * u[0] + (vj[1] - vn[1]) * u[1]
            cnt += 1
        out[n] = acc / cnt if cnt > 0 else 0.0
    return ScalarField(out)

def laplacian(cell: GeometricCell, phi: ScalarField) -> ScalarField:
    """
    Graph Laplacian: Δφ(n) ≈ avg_neighbors φ(j) - φ(n).
    """
    out: Dict[Node, float] = {}
    for n, _ in cell.points.items():
        cnt = 0
        s = 0.0
        for j in neighbors(cell, n):
            s += phi.data.get(j, 0.0)
            cnt += 1
        out[n] = (s / cnt - phi.data.get(n, 0.0)) if cnt > 0 else 0.0
    return ScalarField(out)


# -----------------------------------------------------------------------------
# Energies / potentials
# -----------------------------------------------------------------------------

def kinetic_energy(cell: GeometricCell, v: VectorField, mass: float = 1.0) -> float:
    return 0.5 * mass * sum(vx*vx + vy*vy for (vx, vy) in v.data.values())

def potential_energy(cell: GeometricCell, phi: ScalarField, k: float = 1.0) -> float:
    # Simple quadratic potential ∑ 0.5 k φ^2
    return 0.5 * k * sum(val*val for val in phi.data.values())

def total_energy(cell: GeometricCell, v: VectorField, phi: ScalarField,
                 mass: float = 1.0, k: float = 1.0) -> float:
    return kinetic_energy(cell, v, mass) + potential_energy(cell, phi, k)

# Time integrators
# -----------------------------------------------------------------------------

def euler_step_positions(cell: GeometricCell, v: VectorField, dt: float) -> GeometricCell:
    """
    x_{t+dt} = x_t + v * dt   (node-wise)
    """
    pts = {}
    for n, p in cell.points.items():
        vx, vy = v.data.get(n, (0.0, 0.0))
        pts[n] = Point(p.x + vx * dt, p.y + vy * dt, p.level)
    return GeometricCell(points=pts, edges=set(cell.edges), relations=set(cell.relations))

def euler_step_field(phi: ScalarField, rhs: ScalarField, dt: float) -> ScalarField:
    """
    φ_{t+dt} = φ_t + rhs * dt
    """
    out = {n: phi.data.get(n, 0.0) + rhs.data.get(n, 0.0) * dt
           for n in set(phi.data.keys()) | set(rhs.data.keys())}
    return ScalarField(out)

def semi_implicit_step(
    cell: GeometricCell,
    v: VectorField,
    phi: ScalarField,
    grad_potential: Callable[[GeometricCell, ScalarField], VectorField],
    dt: float,
    mass: float = 1.0
) -> Tuple[GeometricCell, VectorField]:
    """
    Symplectic-ish update:
        v_{t+dt} = v_t - (dt/m) * ∇U(φ_t)
        x_{t+dt} = x_t + dt * v_{t+dt}
    """
    force = grad_potential(cell, phi)  # returns vector field “∇U”
    # velocity update
    new_v = {}
    for n, vec in v.data.items():
        fx, fy = force.data.get(n, (0.0, 0.0))
        new_v[n] = (vec[0] - (dt / mass) * fx, vec[1] - (dt / mass) * fy)
    v2 = VectorField(new_v)
    # position update
    cell2 = euler_step_positions(cell, v2, dt)
    return cell2, v2


# -----------------------------------------------------------------------------
# A sample potential: link-spring energy on edges
# -----------------------------------------------------------------------------

def spring_potential_gradient(cell: GeometricCell, phi: ScalarField,
                              k: float = 1.0, rest: float = 1.0) -> VectorField:
    """
    Interpret φ(n) as a per-node “rest length scale.”
    The potential penalizes stretched/shrunk edges; gradient gives forces.
    """
    forces: Dict[Node, Vec2] = {n: (0.0, 0.0) for n in cell.points}
    for (i, j) in cell.edges:
        pi = cell.points[i]; pj = cell.points[j]
        dx, dy = pj.x - pi.x, pj.y - pi.y
        L = math.hypot(dx, dy) + 1e-9
        u = (dx / L, dy / L)
        # target length uses average of φ(i), φ(j)
        L0 = rest * (0.5 * (phi.data.get(i, 1.0) + phi.data.get(j, 1.0)))
        fmag = k * (L - L0)
        fi = ( fmag * u[0],  fmag * u[1])
        fj = (-fmag * u[0], -fmag * u[1])
        # accumulate
        f_ix, f_iy = forces[i]
        f_jx, f_jy = forces[j]
        forces[i] = (f_ix + fi[0], f_iy + fi[1])
        forces[j] = (f_jx + fj[0], f_jy + fj[1])
    return VectorField(forces)

    return tgt, phi.pullback(lambda X: F.map_obj(X), cell)

def pullback_vector(F, cell: GeometricCell, v: VectorField) -> Tuple[GeometricCell, VectorField]:
    tgt = F.map_obj(cell)
    return tgt, v.pullback(lambda X: F.map_obj(X), cell)

"""
analysis.py — Informational and geometric diagnostics for Relational ∞-Topos Cells

Analyzes output from:

    - physics.py       (field dynamics)

    - Component counts (connected subgraphs)
    - Relational curvature (degree imbalance)
    - Curvature statistics (mean, variance)
    - Coherence index (bounded stability heuristic)

All functions are pure and finite.  Safe to run on any RelationalCell or GeometricCell.
"""

from __future__ import annotations
from typing import Dict, Set, Tuple, Union
import math
from statistics import mean, pstdev
from cell import RelationalCell

Node = Tuple[int, int]
YAdj = Dict[Node, Set[Node]]
RRel = Dict[Node, Set[Node]]

# -----------------------------------------------------------------------------
# --- Core utilities ----------------------------------------------------------
# -----------------------------------------------------------------------------

def nodes_edges_from(obj: Union[RelationalCell, GeometricCell]) -> Tuple[Set[Node], Set[Tuple[Node, Node]]]:
    """Extract node and edge sets uniformly from either cell type."""
    if isinstance(obj, RelationalCell):
        nodes = {n for src, dsts in obj.Y.items() for n in [src, *dsts]}
        edges = {(i,j) for i,js in obj.Y.items() for j in js}
    elif isinstance(obj, GeometricCell):
        nodes = set(obj.points.keys())
        edges = set(obj.edges)
    else:
        raise TypeError("Unsupported cell type")
    return nodes, edges

# -----------------------------------------------------------------------------
# --- Entropy -----------------------------------------------------------------
# -----------------------------------------------------------------------------

def shannon_entropy(obj: Union[RelationalCell, GeometricCell]) -> float:
    """
    Shannon entropy of edge-degree distribution:
        H = -∑_i p_i log₂ p_i,  where p_i = deg(i)/∑_j deg(j)
    """
    nodes, edges = nodes_edges_from(obj)
    if not nodes or not edges:
        return 0.0
    degrees = {n: 0 for n in nodes}
    for i, j in edges:
        degrees[i] = degrees.get(i, 0) + 1
    total = sum(degrees.values())
    if total == 0:
        return 0.0
    p = [deg / total for deg in degrees.values() if deg > 0]
    return -sum(pi * math.log2(pi) for pi in p)

def normalized_entropy(obj: Union[RelationalCell, GeometricCell]) -> float:
    """H / H_max = H / log₂(N)"""
    H = shannon_entropy(obj)
    N = len(nodes_edges_from(obj)[0])
    if N <= 1:
        return 0.0
    return H / math.log2(N)

# -----------------------------------------------------------------------------
# --- Topology ----------------------------------------------------------------
# -----------------------------------------------------------------------------

def euler_characteristic(obj: Union[RelationalCell, GeometricCell]) -> int:
    """χ = |V| - |E|"""
    nodes, edges = nodes_edges_from(obj)
    return len(nodes) - len(edges)

def connected_components(obj: Union[RelationalCell, GeometricCell]) -> int:
    """Return number of connected components (undirected)."""
    nodes, edges = nodes_edges_from(obj)
    if not nodes:
        return 0
    adj: Dict[Node, Set[Node]] = {n: set() for n in nodes}
    for i, j in edges:
        adj[i].add(j)
        adj[j].add(i)
    seen = set()
    comps = 0
    for n in nodes:
        if n in seen:
            continue
        stack = [n]
        while stack:
            x = stack.pop()
            if x in seen: 
                continue
            seen.add(x)
            stack.extend(adj[x])
        comps += 1
    return comps

# -----------------------------------------------------------------------------
# --- Degree & Curvature ------------------------------------------------------
# -----------------------------------------------------------------------------

def degree_statistics(obj: Union[RelationalCell, GeometricCell]) -> Dict[str, float]:
    """Compute mean, stdev, and range of node degrees."""
    nodes, edges = nodes_edges_from(obj)
    deg = {n: 0 for n in nodes}
    for i, j in edges:
        deg[i] += 1
    vals = list(deg.values()) or [0]
    return {
        "mean": mean(vals),
        "stdev": pstdev(vals),
        "min": min(vals),
        "max": max(vals),
    }

def relational_curvature(obj: Union[RelationalCell, GeometricCell]) -> Dict[Node, float]:
    """
    Simple curvature measure: κ(n) = (in_deg - out_deg) / total_deg
    (for GeometricCell edges treated as directed).
    """
    nodes, edges = nodes_edges_from(obj)
    in_deg = {n: 0 for n in nodes}
    out_deg = {n: 0 for n in nodes}
    for i, j in edges:
        out_deg[i] += 1
        in_deg[j] += 1
    curv = {}
    for n in nodes:
        tot = in_deg[n] + out_deg[n]
        curv[n] = (in_deg[n] - out_deg[n]) / tot if tot > 0 else 0.0
    return curv

def curvature_stats(obj: Union[RelationalCell, GeometricCell]) -> Dict[str, float]:
    """Return mean and variance of curvature."""
    curv = list(relational_curvature(obj).values())
    if not curv:
        return {"mean": 0.0, "stdev": 0.0}
    return {"mean": mean(curv), "stdev": pstdev(curv)}

# -----------------------------------------------------------------------------
# --- Coherence ---------------------------------------------------------------
# -----------------------------------------------------------------------------

def coherence_index(obj: Union[RelationalCell, GeometricCell]) -> float:
    """
    Coherence index = normalized_entropy * (1 - |mean_curvature|).
    Heuristic for relational balance: high entropy but balanced flow.
    """
    Hn = normalized_entropy(obj)
    curv = curvature_stats(obj)
    return Hn * (1.0 - abs(curv["mean"]))

def summary(obj: Union[RelationalCell, GeometricCell]) -> str:
    """Return formatted diagnostic summary."""
    H = shannon_entropy(obj)
    Hn = normalized_entropy(obj)
    χ = euler_characteristic(obj)
    comps = connected_components(obj)
    degs = degree_statistics(obj)
    curv = curvature_stats(obj)
    coh = coherence_index(obj)
    return (
        f"Entropy H = {H:.3f}  (norm {Hn:.3f})\n"
        f"Euler χ = {χ},  Components = {comps}\n"
        f"Degree: mean {degs['mean']:.2f}  ±{degs['stdev']:.2f}, "
        f"range [{degs['min']},{degs['max']}]\n"
        f"Curvature: mean {curv['mean']:.3f}  ±{curv['stdev']:.3f}\n"
        f"Coherence index = {coh:.3f}"
    )
