  ;; (elme)
  (require 'ht)

  (setq elme-catalog (ht ('specimens (ht-create))
			 ('locations (ht-create))))

  ;; functions
  (defun elme-add-location! (id info)
    (ht-set! (ht-get elme-catalog 'locations) id info))
  (defun elme-add-specimen! (id info)
    (ht-set! (ht-get elme-catalog 'specimens) id info)
    (elme-get-specimen id))
  (defun elme-get-specimen (id)
    (ht-get (ht-get elme-catalog 'specimens) id))
  (defun elme-get-specimen-attribute (id attr)
    (ht-get (elme-get-specimen id) attr))
  (defun elme-get-specimen-species (id)
    (elme-get-specimen-attribute id 'species))

  ;; data

  (elme-add-location!
