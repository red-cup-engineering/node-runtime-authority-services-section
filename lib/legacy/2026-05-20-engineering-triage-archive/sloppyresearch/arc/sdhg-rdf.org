sdhg:IncidencePathShape
  a sh:NodeShape ;
  sh:targetClass sdhg:RelationalCell ;
  sh:rule [
    a sh:SPARQLRule ;
    sh:prefixes ( sdhg: ) ;
    sh:construct """
      CONSTRUCT { ?edge a sdhg:Edge . } WHERE { ?edge a sdhg:Edge . }
    """ ;
    sh:condition [
      a sh:SPARQLConstraint ;
      sh:message "All exit-paths from incidences must target the terminal face." ;
      sh:select """
        SELECT ?cell WHERE {
          ?cell a sdhg:RelationalCell ;
                sdhg:hasTerminalFace ?t .
          ?cell sdhg:hasIncidence ?e .
          ?e sdhg:admitsExitPath ?p .
          ?p sdhg:targets ?t2 .
          FILTER (?t != ?t2)
        }
      """
    ]
  ] .
